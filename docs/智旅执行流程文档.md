```sql
CREATE DATABASE IF NOT EXISTS smarttravel CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smarttravel;

-- ============================================================
-- 用户表
-- ============================================================
CREATE TABLE IF NOT EXISTS users (
    id VARCHAR(36) PRIMARY KEY,
    nickname VARCHAR(50),
    phone VARCHAR(20) UNIQUE,
    avatar_url VARCHAR(500),
    oauth_provider VARCHAR(20) COMMENT 'wechat / alipay',
    oauth_openid VARCHAR(100),
    oauth_unionid VARCHAR(100),
    password_hash VARCHAR(200),
    last_login_at DATETIME,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_oauth (oauth_provider, oauth_openid),
    INDEX idx_phone (phone)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 用户偏好表
CREATE TABLE IF NOT EXISTS user_preferences (
    id VARCHAR(36) PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL UNIQUE,
    travel_style JSON,
    budget_range JSON,
    constraints JSON,
    favorite_destinations JSON,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 行程表
-- ============================================================
CREATE TABLE IF NOT EXISTS itineraries (
    id VARCHAR(36) PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    title VARCHAR(200) NOT NULL DEFAULT '未命名行程',
    destination VARCHAR(100) NOT NULL,
    start_date DATE,
    end_date DATE,
    days INT NOT NULL,
    total_budget DECIMAL(10, 2),
    status ENUM('draft','planned','in_progress','completed','cancelled') DEFAULT 'draft',
    source ENUM('ai_generated','manual','cloned','replanned') DEFAULT 'ai_generated',
    dify_workflow_run_id VARCHAR(100),
    raw_input JSON,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_status (user_id, status),
    INDEX idx_destination (destination),
    INDEX idx_created (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 行程日表
CREATE TABLE IF NOT EXISTS itinerary_days (
    id VARCHAR(36) PRIMARY KEY,
    itinerary_id VARCHAR(36) NOT NULL,
    day_number INT NOT NULL,
    date DATE,
    weather JSON,
    day_notes TEXT,
    FOREIGN KEY (itinerary_id) REFERENCES itineraries(id) ON DELETE CASCADE,
    UNIQUE KEY uk_itinerary_day (itinerary_id, day_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 日程活动表
CREATE TABLE IF NOT EXISTS day_activities (
    id VARCHAR(36) PRIMARY KEY,
    day_id VARCHAR(36) NOT NULL,
    activity_type ENUM('attraction','hotel','restaurant','transport','other') NOT NULL,
    name VARCHAR(200) NOT NULL,
    address VARCHAR(500),
    latitude DECIMAL(10, 7),
    longitude DECIMAL(10, 7),
    duration_minutes INT DEFAULT 60,
    estimated_cost DECIMAL(10, 2) DEFAULT 0,
    sort_order INT NOT NULL,
    transportation VARCHAR(50),
    travel_time_from_prev INT DEFAULT 0,
    ai_reason TEXT,
    metadata JSON,
    FOREIGN KEY (day_id) REFERENCES itinerary_days(id) ON DELETE CASCADE,
    INDEX idx_day_order (day_id, sort_order)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 行程版本表
CREATE TABLE IF NOT EXISTS itinerary_versions (
    id VARCHAR(36) PRIMARY KEY,
    itinerary_id VARCHAR(36) NOT NULL,
    version_number INT NOT NULL,
    change_description VARCHAR(200),
    snapshot JSON NOT NULL,
    trigger_event VARCHAR(100),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (itinerary_id) REFERENCES itineraries(id) ON DELETE CASCADE,
    UNIQUE KEY uk_version (itinerary_id, version_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 订单表
-- ============================================================
CREATE TABLE IF NOT EXISTS orders (
    id VARCHAR(36) PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    itinerary_id VARCHAR(36),
    status ENUM('pending','paid','timeout','cancelled','refunded') DEFAULT 'pending',
    payment_status ENUM('unpaid','paid','refunding','refunded') DEFAULT 'unpaid',
    total_amount DECIMAL(10, 2) NOT NULL,
    delay_token VARCHAR(100),
    expire_at DATETIME,
    paid_at DATETIME,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_status (user_id, status),
    INDEX idx_expire (expire_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 订单项表
CREATE TABLE IF NOT EXISTS order_items (
    id VARCHAR(36) PRIMARY KEY,
    order_id VARCHAR(36) NOT NULL,
    resource_type ENUM('hotel','ticket','flight','restaurant','insurance') NOT NULL,
    resource_id VARCHAR(100) NOT NULL,
    resource_name VARCHAR(200) NOT NULL,
    unit_price DECIMAL(10, 2) NOT NULL,
    quantity INT DEFAULT 1,
    booking_date DATE,
    check_in DATETIME,
    check_out DATETIME,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    INDEX idx_order (order_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 预订表
-- ============================================================
CREATE TABLE IF NOT EXISTS bookings (
    id VARCHAR(36) PRIMARY KEY,
    itinerary_id VARCHAR(36),
    order_id VARCHAR(36),
    resource_type VARCHAR(50) NOT NULL,
    resource_id VARCHAR(100) NOT NULL,
    status ENUM('reserved','confirmed','cancelled') DEFAULT 'reserved',
    check_in DATETIME,
    check_out DATETIME,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (itinerary_id) REFERENCES itineraries(id) ON DELETE SET NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE SET NULL,

-- ============================================================
-- 审计日志表
-- ============================================================
CREATE TABLE IF NOT EXISTS audit_logs (
    id VARCHAR(36) PRIMARY KEY,

    resource_type VARCHAR(50) NOT NULL,

    INDEX idx_user_time (user_id, created_at),
    INDEX idx_resource (resource_type, resource_id)
```

### Step 2.3 编写 ES 索引初始化脚本

创建 `scripts/init_es.sh`：

```bash
#!/bin/bash
set -e

ES_URL="http://localhost:9200"

echo ">>> Creating ES indices..."

# POI 索引
curl -s -X PUT "$ES_URL/poi" -H 'Content-Type: application/json' -d '{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0,
    "analysis": {
      "analyzer": {
        "ik_max_word_analyzer": { "type": "custom", "tokenizer": "ik_max_word" }
      }
    }
  },
  "mappings": {
    "properties": {
      "poi_id":           { "type": "keyword" },
      "name":             { "type": "text", "analyzer": "ik_max_word", "search_analyzer": "ik_smart" },
      "name_suggest":     { "type": "completion" },
      "type":             { "type": "keyword" },
      "city":             { "type": "keyword" },
      "district":         { "type": "keyword" },
      "location":         { "type": "geo_point" },
      "price_range":      { "type": "integer_range" },
      "rating":           { "type": "float" },
      "tags":             { "type": "keyword" },
      "description":      { "type": "text", "analyzer": "ik_max_word" },
      "opening_hours":    { "type": "keyword" },
      "image_url":        { "type": "keyword", "index": false },
      "popularity_score": { "type": "float" },
      "created_at":       { "type": "date" },
      "updated_at":       { "type": "date" }
    }
  }
}'

# 行程模板索引
curl -s -X PUT "$ES_URL/itinerary_templates" -H 'Content-Type: application/json' -d '{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0
  },
  "mappings": {
    "properties": {
      "itinerary_id":    { "type": "keyword" },
      "title":           { "type": "text", "analyzer": "ik_max_word" },
      "destination":     { "type": "keyword" },
      "days":            { "type": "integer" },
      "budget":          { "type": "float" },
      "tags":            { "type": "keyword" },
      "travel_style":    { "type": "keyword" },
      "season":          { "type": "keyword" },
      "likes_count":     { "type": "integer" },
      "copies_count":    { "type": "integer" },
      "user_id":         { "type": "keyword" },
      "created_at":      { "type": "date" }
    }
  }
}'

echo ">>> ES indices created successfully!"
echo ">>> Verify: curl $ES_URL/_cat/indices"
```

**关键代码 `backend/common/common/auth/jwt.py`**：

```python
from datetime import datetime, timedelta, timezone
from jose import jwt, JWTError
import uuid

SECRET_KEY = "smarttravel-secret-key-change-in-production-2026"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 15
REFRESH_TOKEN_EXPIRE_DAYS = 7


def create_access_token(user_id: str) -> str:
    now = datetime.now(timezone.utc)
    expire = now + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    return jwt.encode(
        {
            "sub": user_id,
            "exp": expire,
            "iat": now,
            "type": "access",
            "jti": str(uuid.uuid4()),
        },
        SECRET_KEY,
        algorithm=ALGORITHM,
    )


def create_refresh_token(user_id: str) -> str:
    now = datetime.now(timezone.utc)
    expire = now + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)
    return jwt.encode(
        {
            "sub": user_id,
            "exp": expire,
            "iat": now,
            "type": "refresh",
            "jti": str(uuid.uuid4()),
        },
        SECRET_KEY,
        algorithm=ALGORITHM,
    )


def decode_token(token: str) -> dict:
    try:
        return jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
    except JWTError:
        raise ValueError("Invalid or expired token")
```

```python
import bcrypt
from uuid import uuid4
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.models.user import User
from common.auth.jwt import create_access_token, create_refresh_token
from common.redis_client import get_redis


class AuthService:
    async def register(self, db: AsyncSession, phone: str, password: str) -> dict:
        """手机号注册"""
        # 检查手机号是否已注册
        existing = await db.execute(select(User).where(User.phone == phone))
        if existing.scalar_one_or_none():
            raise ValueError("Phone already registered")
        
        # 创建用户
        user = User(
            id=str(uuid4()),
            phone=phone,
            password_hash=bcrypt.hashpw(
                password.encode(), bcrypt.gensalt()
            ).decode(),
        )
        db.add(user)
        await db.commit()
        await db.refresh(user)
        
        return {
            "user_id": user.id,
            "access_token": create_access_token(user.id),
            "refresh_token": create_refresh_token(user.id),
        }

    async def login(self, db: AsyncSession, phone: str, password: str) -> dict:
        """手机号登录"""
        result = await db.execute(select(User).where(User.phone == phone))
        user = result.scalar_one_or_none()
        if not user:
            raise ValueError("User not found")
        
        if not bcrypt.checkpw(password.encode(), user.password_hash.encode()):
            raise ValueError("Invalid password")
        
        return {
            "user_id": user.id,
            "access_token": create_access_token(user.id),
            "refresh_token": create_refresh_token(user.id),
        }

    async def logout(self, user_id: str, jti: str):
        """登出：将JTI加入Redis黑名单"""
        redis = get_redis()
        await redis.setex(f"token:blacklist:{jti}", 900, "1")
```

```python
from elasticsearch import AsyncElasticsearch
from typing import Optional


class ESService:
    def __init__(self, es_url: str):
        self.client = AsyncElasticsearch(es_url)

    async def search_poi(
        self,
        keyword: str,
        city: Optional[str] = None,
        poi_type: Optional[str] = None,
        lat: Optional[float] = None,
        lon: Optional[float] = None,
        distance: Optional[str] = None,
        min_price: Optional[int] = None,
        max_price: Optional[int] = None,
        tags: Optional[list] = None,
        page: int = 1,
        page_size: int = 20,
    ) -> dict:
        """多维度POI搜索"""
        must = []
        if keyword:
            must.append({
                "multi_match": {
                    "query": keyword,
                    "fields": ["name^3", "description", "tags"],
                }
            })
        if city:
            must.append({"term": {"city": city}})
        if poi_type:
            must.append({"term": {"type": poi_type}})

        filters = []
        if lat is not None and lon is not None and distance:
            filters.append({
                "geo_distance": {
                    "distance": distance,
                    "location": {"lat": lat, "lon": lon},
                }
            })
        if min_price is not None or max_price is not None:
            price_range = {}
            if min_price is not None:
                price_range["gte"] = min_price
            if max_price is not None:
                price_range["lte"] = max_price
            filters.append({"range": {"price_range": price_range}})

        body = {
            "query": {
                "bool": {
                    "must": must,
                    "filter": filters,
                }
            },
            "from": (page - 1) * page_size,
            "size": page_size,
        }

        result = await self.client.search(index="poi", body=body)
        total = result["hits"]["total"]["value"]
        hits = [hit["_source"] for hit in result["hits"]["hits"]]
        return {"total": total, "data": hits, "page": page, "page_size": page_size}

    async def suggest(self, prefix: str) -> list:
        """搜索自动补全"""
        result = await self.client.search(
            index="poi",
            body={
                "suggest": {
                    "poi_suggest": {
                        "prefix": prefix,
                        "completion": {"field": "name_suggest"},
                    }
                }
            },
        )
        suggestions = result["suggest"]["poi_suggest"][0]["options"]
        return [s["_source"]["name"] for s in suggestions]
```

```python
    )

    redis = get_redis()
    for item in items:
        lock_key = f"resource:lock:{item['resource_type']}:{item['resource_id']}"
        await redis.setex(lock_key, 900, order_id)  # 15分钟TTL
    
    # 创建订单项

            id=str(uuid4()),
            order_id=order_id,
            resource_type=item_data["resource_type"],
            resource_id=item_data["resource_id"],
            resource_name=item_data["resource_name"],
            unit_price=item_data["unit_price"],
            quantity=item_data.get("quantity", 1),
        )
        db.add(item)
    await db.commit()
    
    # TODO: 投递RabbitMQ延迟消息（15分钟后检查）

async def confirm_payment(db: AsyncSession, order_id: str) -> Order:
    """确认支付"""
    order = await db.get(Order, order_id)
    if not order or order.status != "pending":

    if datetime.now(timezone.utc) > order.expire_at:
        order.status = "timeout"
        await db.commit()

    order.status = "paid"
    order.payment_status = "paid"
    order.paid_at = datetime.now(timezone.utc)
    await db.commit()
```

**SSE连接Composable `frontend/src/composables/useItineraryGenerate.ts`**：

```typescript
import { ref } from 'vue'
import { useAuthStore } from '@/stores/auth'

interface SSEEvent {
  event: string
  data: any
}

export function useItineraryGenerate() {
  const messages = ref<any[]>([])
  const currentItinerary = ref<any>(null)
  const isGenerating = ref(false)
  let abortController: AbortController | null = null

  async function generate(message: string) {
    isGenerating.value = true
    abortController = new AbortController()
    const authStore = useAuthStore()

    try {
      const response = await fetch('/api/v1/itinerary/generate', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${authStore.accessToken}`,
        },
        body: JSON.stringify({ message }),
        signal: abortController.signal,
      })

      const reader = response.body!.getReader()
      const decoder = new TextDecoder()
      let buffer = ''

      while (true) {
        const { done, value } = await reader.read()
        if (done) break

        buffer += decoder.decode(value, { stream: true })
        const lines = buffer.split('\n')
        buffer = lines.pop() || ''

        for (const line of lines) {
          if (line.startsWith('event: ')) {
            // 暂存事件类型，等待data行
            continue
          }
          if (line.startsWith('data: ')) {
            try {
              const data = JSON.parse(line.slice(6))
              messages.value.push(data)
              
              // 处理不同类型的事件
              if (data.event === 'intent_result') {
                messages.value.push({ type: 'intent', content: data })
              } else if (data.event === 'day_result') {
                // 更新行程预览
                if (!currentItinerary.value) {
                  currentItinerary.value = { days: [] }
                }
                currentItinerary.value.days[data.day - 1] = data
              } else if (data.event === 'complete') {
                currentItinerary.value = data
              }
            } catch (e) {
              // 非JSON行，忽略
            }
          }
        }
      }
    } finally {
      isGenerating.value = false
    }
  }

  function abort() {
    abortController?.abort()
    isGenerating.value = false
  }

  return { messages, currentItinerary, isGenerating, generate, abort }
}
```

```typescript
import { ref, onUnmounted } from 'vue'
import { useAuthStore } from '@/stores/auth'

export function useWebSocket() {
  const ws = ref<WebSocket | null>(null)
  const replanData = ref<any>(null)
  const isConnected = ref(false)

  function connect() {
    const authStore = useAuthStore()
    ws.value = new WebSocket(`ws://localhost:8005/ws/connect?token=${authStore.accessToken}`)

    ws.value.onopen = () => { isConnected.value = true }
    ws.value.onmessage = (event) => {
      const data = JSON.parse(event.data)
      if (data.type === 'replan_ready') {
        replanData.value = data
      }
    }
    ws.value.onclose = () => {
      isConnected.value = false
      setTimeout(connect, 3000) // 自动重连
    }
  }

  function disconnect() {
    ws.value?.close()
  }

  connect()
  onUnmounted(disconnect)

  return { replanData, isConnected, connect, disconnect }
}
```

## 执行铁律

> 以下3条每天开工前读一遍：

### 铁律1：先跑通，再完善
不是"先做完美的架构再写代码"，而是：
```
一个Service + 一个API + 一个SQL → 立刻curl验证 → 通过后再做下一个
```

### 铁律2：核心链路优先
Dify行程生成是命门。如果W2结束它还没通：
- 砍掉ES搜索
- 砍掉支付（先做静态数据）
- 砍掉动态调整（留MQ骨架即可）
- **只保留：登录 → AI生成行程 → 编辑器 → 保存**

### 铁律3：AI是你的副驾驶
- CRUD代码 → AI写（你审查）
- 前端组件 → AI写（你调整）
- SQL建表 → AI生成（你检查索引）
- **但以下必须亲手做**：
  - Dify Prompt设计（面试官会深问这个）
  - 微服务拆分边界（为什么这样分？）
  - ES索引Mapping设计（为什么这些字段？）
  - RabbitMQ拓扑设计（为什么用Topic而不是Direct？）
  - 延迟支付时序设计（并发安全吗？）
